#!/usr/bin/env node
import fs from 'node:fs';
import path from 'node:path';
import assert from 'node:assert/strict';
import { fileURLToPath } from 'node:url';

const here = path.dirname(fileURLToPath(import.meta.url));
const root = path.resolve(here, '..');
const read = (p) => JSON.parse(fs.readFileSync(path.join(root,p),'utf8'));

const manifest = read('manifest.json');
const stages = read('process/stages.json');
const al = read('objects/al-struct-1.schema.json');
const validators = read('execution/validators.json');
const context = read('policy/context-policy.json');
const tests = read('runtime/contract-tests.json');

assert.equal(manifest.bundle_version, '2.2.0');
assert.equal(manifest.semantic_contract, 'AL-STRUCT-1');
assert.equal(stages.length, 30);

const required = al.required;
assert.deepEqual(required, ['passport','outputs','annotations','trace','input_fate','changes','completion']);
assert.equal(al.properties.passport.properties.contract.const, 'AL-STRUCT-1');
assert.deepEqual(al.properties.outputs.items.properties.type.enum, ['ANSWER','QUESTION','REJECT']);
assert.ok(al.properties.annotations.items.properties.type.enum.includes('VERIFIED'));
assert.deepEqual(al.properties.input_fate.items.properties.disposition.enum,
  ['CONSUMED','PRESERVED','TRANSFORMED','REJECTED','SUPERSEDED','NOT_USED']);
assert.deepEqual(al.properties.changes.items.properties.op.enum, ['CREATE','UPDATE','SUPERSEDE','MERGE']);
assert.deepEqual(al.properties.completion.properties.status.enum, ['COMPLETE','PARTIAL','FAILED']);

for (const v of [
  'al_struct_1_valid','al_struct_internal_consistency','snapshot_hash_ack_valid',
  'reference_first_valid','reserved_runtime_fields_absent','empty_by_design_valid',
  'context_budget_valid','source_message_idempotent','context_audit_reproducible','role_no_arbitration'
]) assert.ok(validators[v], `missing validator ${v}`);

assert.equal(context.maxPromptChars, 60000);
assert.equal(context.maxOutputContentChars, 8000);
assert.deepEqual(context.assembly_order, ['RULES','STATE','ACTIVE','DELTA','TASK']);
assert.equal(context.safe_truncation.marker, '[OBJ:TRUNC]');
assert.equal(context.silent_eviction, false);

const ids = new Set(tests.map(t => t.id));
for (const id of [
  'AL_STRUCT_REQUIRED_FIELDS','SNAPSHOT_HASH_MISMATCH_REJECTED',
  'SOURCE_MESSAGE_DUPLICATE_SUPPRESSED','CONTEXT_ASSEMBLY_AUDIT_REPRODUCIBLE',
  'ROLE_DOES_NOT_GRANT_ARBITRATION'
]) assert.ok(ids.has(id), `missing contract test ${id}`);

for (const s of stages) {
  assert.ok(s.n >= 1 && s.n <= 30);
  if (s.response_contract?.profile !== 'DETERMINISTIC') {
    assert.equal(s.response_contract.model_output_schema, 'AL_STRUCT_1');
    assert.equal(s.response_contract.semantic_contract, 'AL-STRUCT-1');
    assert.equal(s.response_contract.snapshot_ack_required, true);
  }
}

console.log(`PASS: ${stages.length} stages, ${tests.length} contract tests, AL-STRUCT-1 + context policy v2.2 static integrity.`);
